calc_metrics <- function(observed, fitted) {
  
  ok <- is.finite(observed) & is.finite(fitted)
  observed <- observed[ok]
  fitted <- fitted[ok]
  
  if (length(observed) < 5) {
    return(data.table(
      CCC = NA_real_,
      RMSE = NA_real_,
      MAE = NA_real_,
      Bias = NA_real_,
      R2 = NA_real_,
      Residual_SD = NA_real_,
      n_eval = length(observed)
    ))
  }
  
  resid <- observed - fitted
  
  ccc_val <- tryCatch(
    DescTools::CCC(observed, fitted)$rho.c[, "est"],
    error = function(e) NA_real_
  )
  
  data.table(
    CCC = as.numeric(ccc_val),
    RMSE = sqrt(mean(resid^2, na.rm = TRUE)),
    MAE = mean(abs(resid), na.rm = TRUE),
    Bias = mean(fitted - observed, na.rm = TRUE),
    R2 = cor(observed, fitted, use = "complete.obs")^2,
    Residual_SD = sd(resid, na.rm = TRUE),
    n_eval = length(observed)
  )
}

sample_cells_region <- function(cell_table,

                                n_cells_per_stratum = 250,
                                seed = NULL) {

extract_year_effects <- function(model, region_name, run_id, response_var) {
  
  ran <- ranef(model)
  year_name <- grep("Ano", names(ran), value = TRUE)[1]
  
  if (is.na(year_name)) return(NULL)
  
  year_eff <- as.data.frame(ran[[year_name]])
  year_eff$Ano <- rownames(year_eff)
  names(year_eff)[1] <- "year_effect"
  
  year_eff %>%
    mutate(
      region = region_name,
      response = response_var,
      run = run_id,
      Ano = as.numeric(as.character(Ano)),
      year_effect = as.numeric(year_effect)
    )
}

make_spatial_component_fast <- function(model,
                                        dados_sub,
                                        fitted_full,
                                        fixed_part,
                                        region_name,
                                        run_id,
                                        response_var,
                                        max_points = 6000) {
  
  ran <- ranef(model)
  year_name <- grep("Ano", names(ran), value = TRUE)[1]
  
  if (is.na(year_name)) {
    dados_sub$year_effect <- 0
  } else {
    year_eff_df <- as.data.frame(ran[[year_name]])
    year_eff_df$Ano <- rownames(year_eff_df)
    names(year_eff_df)[1] <- "year_effect"
    
    year_eff_df <- year_eff_df %>%
      mutate(
        Ano = as.character(Ano),
        year_effect = as.numeric(year_effect)
      )
    
    dados_sub <- dados_sub %>%
      mutate(Ano_chr = as.character(Ano)) %>%
      left_join(year_eff_df, by = c("Ano_chr" = "Ano")) %>%
      mutate(year_effect = ifelse(is.na(year_effect), 0, year_effect))
  }
  
  tmp <- dados_sub %>%
    mutate(
      fitted_full = fitted_full,
      fixed_part = fixed_part,
      spatial_component = fitted_full - fixed_part - year_effect
    )
  
  spatial_cell <- tmp %>%
    group_by(cell_id, Lon, Lat) %>%
    summarise(
      spatial_component = mean(spatial_component, na.rm = TRUE),
      .groups = "drop"
    ) %>%
    mutate(
      region = region_name,
      response = response_var,
      run = run_id
    )
  
  if (nrow(spatial_cell) > max_points) {
    spatial_cell <- spatial_cell %>%
      slice_sample(n = max_points)
  }
  
  spatial_cell
}

run_one_spamm_loaded <- function(region_name,
                                 run_id,
                                 data_region,
                                 cell_table,
                                 out_dir,
                                 response_var,
                                 n_cells_per_stratum = 250,
                                 df_frac = 3,
                                 seed = NULL,
                                 save_model = TRUE,
                                 save_run_data = TRUE,
                                 diagnostic_sample_max = 30000) {
  
  spaMM.options(nb_cores = 1)
  
  run_chr <- sprintf("%03d", run_id)
  
  file_done <- file.path(out_dir, paste0("DONE_run_", run_chr, "_", region_name, ".txt"))
  file_model <- file.path(out_dir, "models", paste0("model_run_", run_chr, "_", region_name, ".rds"))
  file_data <- file.path(out_dir, "run_data", paste0("data_run_", run_chr, "_", region_name, ".rds"))
  file_contrasts <- file.path(out_dir, "contrasts", paste0("contrasts_run_", run_chr, "_", region_name, ".csv"))
  file_year <- file.path(out_dir, "year_effects", paste0("year_effects_run_", run_chr, "_", region_name, ".csv"))
  file_spatial <- file.path(out_dir, "spatial_component", paste0("spatial_component_run_", run_chr, "_", region_name, ".csv"))
  file_diag <- file.path(out_dir, "diagnostics", paste0("diagnostics_run_", run_chr, "_", region_name, ".csv"))
  file_metrics <- file.path(out_dir, "metrics", paste0("metrics_run_", run_chr, "_", region_name, ".csv"))
  file_log <- file.path(out_dir, "logs", paste0("log_run_", run_chr, "_", region_name, ".csv"))
  
  if (
    file.exists(file_done) &&
    file.exists(file_contrasts) &&
    file.exists(file_year) &&
    file.exists(file_spatial) &&
    file.exists(file_diag) &&
    file.exists(file_metrics)
  ) {
    message("Skipping completed run ", run_id, " | ", region_name)
    return(NULL)
  }
  
  message("\n===================================")
  message("Running run ", run_id, " | region ", region_name)
  message("===================================")
  
  t0 <- Sys.time()
  
  log_status <- "success"
  log_error <- NA_character_
  
  result <- tryCatch({
    
    selected_cells <- sample_cells_region(
      cell_table = cell_table,
      region_name = region_name,
      n_cells_per_stratum = n_cells_per_stratum,
      seed = seed
    )
    
    dados_sub <- as.data.table(data_region)[
      cell_id %in% selected_cells,
      .(
        response,
        cell_id,
        Ano,
        Tipo,
        frac_area,
        X_km_s,
        Y_km_s,
        Lon,
        Lat
      )
    ]
    
    dados_sub <- droplevels(as.data.frame(dados_sub))
    
    form <- as.formula(
      paste0(
        "response ~ Tipo * ns(frac_area, df = ", df_frac, ") + ",
        "(1 | Ano) + ",
        "Matern(1 | X_km_s + Y_km_s)"
      )
    )
    
    model <- fitme(
      form,
      data = dados_sub,
      family = gaussian(),
      method = "REML"
    )
    
    if (save_model) {
      saveRDS(model, file_model)
    }
    
    if (save_run_data) {
      saveRDS(dados_sub, file_data)
    }
    
    fixed_form <- as.formula(
      paste0("~ Tipo * ns(frac_area, df = ", df_frac, ")")
    )
    
    beta <- fixef(model)
    
    
    # Spatially adjusted contrasts
    frac_seq <- seq(0.01, 1.00, length.out = 50)
    
    newdata_adj <- expand.grid(
      frac_area = frac_seq,
      Tipo = levels(dados_sub$Tipo),
      KEEP.OUT.ATTRS = FALSE,
      stringsAsFactors = FALSE
    )
    
    newdata_adj$Tipo <- factor(
      newdata_adj$Tipo,
      levels = levels(dados_sub$Tipo)
    )
    
    X_adj <- model.matrix(fixed_form, data = newdata_adj)
    common_adj <- intersect(colnames(X_adj), names(beta))
    
    newdata_adj$pred <- as.numeric(
      X_adj[, common_adj, drop = FALSE] %*% beta[common_adj]
    )
    
    contrasts_adj <- newdata_adj %>%
      select(frac_area, Tipo, pred) %>%
      pivot_wider(names_from = Tipo, values_from = pred) %>%
      mutate(
        region = region_name,
        response = response_var,
        run = run_id,
        contrast_type = "Spatially adjusted",
        PF_minus_SF_early = PF - SF_early,
        PF_minus_SF_intermediate = PF - SF_intermediate,
        PF_minus_SF_advanced = PF - SF_advanced
      ) %>%
      select(
        region, response, run, contrast_type, frac_area,
        PF_minus_SF_early,
        PF_minus_SF_intermediate,
        PF_minus_SF_advanced
      )
    

    # Full prediction, residuals and diagnostics
    fitted_full <- as.numeric(
      predict(model, newdata = dados_sub, type = "response")
    )
    
    residual_model <- dados_sub$response - fitted_full
    
    diagnostics <- data.table(
      region = region_name,
      response = response_var,
      run = run_id,
      observed = dados_sub$response,
      fitted = fitted_full,
      residual = residual_model,
      Tipo = as.character(dados_sub$Tipo),
      frac_area = dados_sub$frac_area,
      Ano = as.character(dados_sub$Ano),
      cell_id = dados_sub$cell_id,
      Lon = dados_sub$Lon,
      Lat = dados_sub$Lat
    )
    
    if (nrow(diagnostics) > diagnostic_sample_max) {
      set.seed(seed + 999)
      diagnostics_save <- diagnostics[
        sample(.N, diagnostic_sample_max)
      ]
    } else {
      diagnostics_save <- diagnostics
    }
    
    fwrite(diagnostics_save, file_diag)
    
    metrics <- calc_metrics(
      observed = diagnostics$observed,
      fitted = diagnostics$fitted
    )
    
    metrics[, `:=`(
      region = region_name,
      response = response_var,
      run = run_id,
      n_cells_per_stratum = n_cells_per_stratum,
      selected_cells = length(selected_cells),
      rows_model = nrow(dados_sub),
      n_unique_cells = dplyr::n_distinct(dados_sub$cell_id)
    )]
    
    setcolorder(
      metrics,
      c(
        "region", "response", "run",
        "CCC", "RMSE", "MAE", "Bias", "R2", "Residual_SD", "n_eval",
        "n_cells_per_stratum", "selected_cells", "rows_model", "n_unique_cells"
      )
    )
    
    fwrite(metrics, file_metrics)
    

    # Current landscape contrasts
    dados_pred <- dados_sub %>%
      mutate(
        pred_full = fitted_full,
        frac_area_bin = cut(
          frac_area,
          breaks = seq(0, 1, by = 0.05),
          include.lowest = TRUE
        )
      )
    
    landscape_mean <- dados_pred %>%
      group_by(frac_area_bin, Tipo) %>%
      summarise(
        frac_area = median(frac_area, na.rm = TRUE),
        pred = mean(pred_full, na.rm = TRUE),
        .groups = "drop"
      )
    
    pf_landscape <- landscape_mean %>%
      filter(Tipo == "PF") %>%
      dplyr::select(frac_area_bin, frac_area, PF = pred)
    
    sf_landscape <- landscape_mean %>%
      filter(Tipo != "PF") %>%
      dplyr::select(frac_area_bin, Tipo, pred) %>%
      pivot_wider(names_from = Tipo, values_from = pred)
    
    for (nm in c("SF_early", "SF_intermediate", "SF_advanced")) {
      if (!nm %in% names(sf_landscape)) sf_landscape[[nm]] <- NA_real_
    }
    
    contrasts_landscape <- pf_landscape %>%
      left_join(sf_landscape, by = "frac_area_bin") %>%
      mutate(
        region = region_name,
        response = response_var,
        run = run_id,
        contrast_type = "Current landscape",
        PF_minus_SF_early = PF - SF_early,
        PF_minus_SF_intermediate = PF - SF_intermediate,
        PF_minus_SF_advanced = PF - SF_advanced
      ) %>%
      dplyr::select(
        region, response, run, contrast_type, frac_area,
        PF_minus_SF_early,
        PF_minus_SF_intermediate,
        PF_minus_SF_advanced
      )
    
    contrasts <- bind_rows(
      contrasts_adj,
      contrasts_landscape
    )
    
    fwrite(contrasts, file_contrasts)
    

    # Year and spatial components
    year_effects <- extract_year_effects(
      model = model,
      region_name = region_name,
      run_id = run_id,
      response_var = response_var
    )
    
    fwrite(year_effects, file_year)
    
    X_sub <- model.matrix(fixed_form, data = dados_sub)
    common_sub <- intersect(colnames(X_sub), names(beta))
    
    fixed_part <- as.numeric(
      X_sub[, common_sub, drop = FALSE] %*% beta[common_sub]
    )
    
    spatial_component <- make_spatial_component_fast(
      model = model,
      dados_sub = dados_sub,
      fitted_full = fitted_full,
      fixed_part = fixed_part,
      region_name = region_name,
      run_id = run_id,
      response_var = response_var,
      max_points = 6000
    )
    
    fwrite(spatial_component, file_spatial)
    
    writeLines("done", file_done)
    
    invisible(TRUE)
    
  }, error = function(e) {
    
    log_status <<- "error"
    log_error <<- conditionMessage(e)
    message("ERROR in run ", run_id, " | ", region_name, ": ", log_error)
    invisible(FALSE)
  })
  
  elapsed_min <- round(
    as.numeric(difftime(Sys.time(), t0, units = "mins")),
    2
  )
  
  fwrite(
    data.table(
      region = region_name,
      response = response_var,
      run = run_id,
      status = log_status,
      error_message = log_error,
      n_cells_per_stratum = n_cells_per_stratum,
      df_frac = df_frac,
      elapsed_min = elapsed_min
    ),
    file_log
  )
  
  message("Finished run ", run_id, " | status: ", log_status, " | ", elapsed_min, " min")
  
  invisible(gc())
  NULL
}

load_response_results <- function(response_var, region_name) {
  
  out_dir <- get_out_dir(response_var, region_name)
  
  contrasts_ensemble <- read_all_files(
  file.path(out_dir, "contrasts"),
  "^contrasts_run_.*\\.csv$"
  )
  
  year_effects_ensemble <- read_all_files(
    file.path(out_dir, "year_effects"),
    "^year_effects_run_.*\\.csv$"
  )
  
  spatial_component_ensemble <- read_all_files(
    file.path(out_dir, "spatial_component"),
    "^spatial_component_run_.*\\.csv$"
  )
  
  diagnostics_ensemble <- read_all_files(
    file.path(out_dir, "diagnostics"),
    "^diagnostics_run_.*\\.csv$"
  )
  
  metrics_ensemble <- read_all_files(
    file.path(out_dir, "metrics"),
    "^metrics_run_.*\\.csv$"
  )
  
  logs_ensemble <- read_all_files(
    file.path(out_dir, "logs"),
    "^log_run_.*\\.csv$"
  )
  
  fwrite(contrasts_ensemble, file.path(out_dir, "tables", "contrasts_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(year_effects_ensemble, file.path(out_dir, "tables", "year_effects_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(spatial_component_ensemble, file.path(out_dir, "tables", "spatial_component_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(diagnostics_ensemble, file.path(out_dir, "tables", "diagnostics_ensemble_sampled_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(metrics_ensemble, file.path(out_dir, "tables", "metrics_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(logs_ensemble, file.path(out_dir, "tables", "logs_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  
  contrasts_long <- contrasts_ensemble %>%
  pivot_longer(
    cols = starts_with("PF_minus_SF"),
    names_to = "stage",
    values_to = "difference"
  ) %>%
  mutate(
    stage = recode(
      stage,
      "PF_minus_SF_early" = "SF[e]",
      "PF_minus_SF_intermediate" = "SF[i]",
      "PF_minus_SF_advanced" = "SF[a]"
    ),
    stage = factor(stage, levels = c("SF[e]", "SF[i]", "SF[a]")),
    contrast_type = factor(
      contrast_type,
      levels = c("Spatially adjusted", "Current landscape")
    ),
    frac_bin = case_when(
      contrast_type == "Current landscape" ~ as.character(cut(
        frac_area,
        breaks = seq(0, 1, by = 0.05),
        labels = bin_levels,
        include.lowest = TRUE,
        right = TRUE
      )),
      TRUE ~ NA_character_
    ),
    frac_bin = factor(frac_bin, levels = bin_levels),
    frac_plot = case_when(
      contrast_type == "Spatially adjusted" ~ round(frac_area, 6),
      contrast_type == "Current landscape" ~ as.numeric(frac_bin)
    ),
    response = response_var
  )
  
  contrasts_summary <- contrasts_long %>%
    filter(!is.na(frac_plot), !is.na(difference)) %>%
    group_by(region, response, contrast_type, frac_plot, frac_bin, stage) %>%
    summarise(
      median = median(difference, na.rm = TRUE),
      lower = safe_quantile(difference, 0.025),
      upper = safe_quantile(difference, 0.975),
      n_runs = n_distinct(run),
      .groups = "drop"
    )
  
  fwrite(
  contrasts_summary,
  file.path(out_dir, "tables", "contrasts_summary_empirical_CI.csv"),  sep = ";",  dec = ",")
  
  year_summary <- year_effects_ensemble %>%
    mutate(response = response_var) %>%
    group_by(region, response, Ano) %>%
    summarise(
      median = median(year_effect, na.rm = TRUE),
      lower = safe_quantile(year_effect, 0.025),
      upper = safe_quantile(year_effect, 0.975),
      n_runs = sum(!is.na(year_effect)),
      .groups = "drop"
    )
  
  fwrite(
  year_summary,
  file.path(out_dir, "tables", "year_effects_summary_empirical_CI.csv"),  sep = ";",  dec = ",")
  
  spatial_summary <- spatial_component_ensemble %>%
    mutate(response = response_var) %>%
    group_by(region, response, Lon, Lat) %>%
    summarise(
      spatial_median = median(spatial_component, na.rm = TRUE),
      spatial_mean = mean(spatial_component, na.rm = TRUE),
      spatial_sd = sd(spatial_component, na.rm = TRUE),
      n_runs = n(),
      .groups = "drop"
    )
  
  fwrite(
  spatial_summary,
  file.path(out_dir, "tables", "spatial_component_summary.csv"),  sep = ";",  dec = ",")
  
  metrics_summary <- metrics_ensemble %>%
    group_by(region, response) %>%
    summarise(
      n_successful_runs = n(),
      
      CCC_median = median(CCC, na.rm = TRUE),
      CCC_lower = safe_quantile(CCC, 0.025),
      CCC_upper = safe_quantile(CCC, 0.975),
      
      RMSE_median = median(RMSE, na.rm = TRUE),
      RMSE_lower = safe_quantile(RMSE, 0.025),
      RMSE_upper = safe_quantile(RMSE, 0.975),
      
      MAE_median = median(MAE, na.rm = TRUE),
      MAE_lower = safe_quantile(MAE, 0.025),
      MAE_upper = safe_quantile(MAE, 0.975),
      
      Bias_median = median(Bias, na.rm = TRUE),
      Bias_lower = safe_quantile(Bias, 0.025),
      Bias_upper = safe_quantile(Bias, 0.975),
      
      R2_median = median(R2, na.rm = TRUE),
      R2_lower = safe_quantile(R2, 0.025),
      R2_upper = safe_quantile(R2, 0.975),
      
      .groups = "drop"
    )

  fwrite(
    metrics_summary,
    file.path(out_dir, "tables", "metrics_summary_empirical_CI.csv"),  sep = ";",  dec = ",")

  
  list(
    contrasts_summary = contrasts_summary,
    year_summary = year_summary,
    spatial_summary = spatial_summary
  )
}

safe_quantile <- function(x, p) {
  if (all(is.na(x))) return(NA_real_)
  as.numeric(quantile(x, p, na.rm = TRUE))
}

read_all_files <- function(path, pattern) {
  files <- list.files(path, pattern = pattern, full.names = TRUE)
  if (length(files) == 0) return(data.table())
  rbindlist(lapply(files, fread), fill = TRUE)
}

load_response_results <- function(response_var, region_name) {
  
  out_dir <- get_out_dir(response_var, region_name)
  
  contrasts_ensemble <- read_all_files(
  file.path(out_dir, "contrasts"),
  "^contrasts_run_.*\\.csv$"
  )
  
  year_effects_ensemble <- read_all_files(
    file.path(out_dir, "year_effects"),
    "^year_effects_run_.*\\.csv$"
  )
  
  spatial_component_ensemble <- read_all_files(
    file.path(out_dir, "spatial_component"),
    "^spatial_component_run_.*\\.csv$"
  )
  
  diagnostics_ensemble <- read_all_files(
    file.path(out_dir, "diagnostics"),
    "^diagnostics_run_.*\\.csv$"
  )
  
  metrics_ensemble <- read_all_files(
    file.path(out_dir, "metrics"),
    "^metrics_run_.*\\.csv$"
  )
  
  logs_ensemble <- read_all_files(
    file.path(out_dir, "logs"),
    "^log_run_.*\\.csv$"
  )
  
  fwrite(contrasts_ensemble, file.path(out_dir, "tables", "contrasts_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(year_effects_ensemble, file.path(out_dir, "tables", "year_effects_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(spatial_component_ensemble, file.path(out_dir, "tables", "spatial_component_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(diagnostics_ensemble, file.path(out_dir, "tables", "diagnostics_ensemble_sampled_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(metrics_ensemble, file.path(out_dir, "tables", "metrics_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  fwrite(logs_ensemble, file.path(out_dir, "tables", "logs_ensemble_all_runs.csv"),  sep = ";",  dec = ",")
  
  contrasts_long <- contrasts_ensemble %>%
  pivot_longer(
    cols = starts_with("PF_minus_SF"),
    names_to = "stage",
    values_to = "difference"
  ) %>%
  mutate(
    stage = recode(
      stage,
      "PF_minus_SF_early" = "SF[e]",
      "PF_minus_SF_intermediate" = "SF[i]",
      "PF_minus_SF_advanced" = "SF[a]"
    ),
    stage = factor(stage, levels = c("SF[e]", "SF[i]", "SF[a]")),
    contrast_type = factor(
      contrast_type,
      levels = c("Spatially adjusted", "Current landscape")
    ),
    frac_bin = case_when(
      contrast_type == "Current landscape" ~ as.character(cut(
        frac_area,
        breaks = seq(0, 1, by = 0.05),
        labels = bin_levels,
        include.lowest = TRUE,
        right = TRUE
      )),
      TRUE ~ NA_character_
    ),
    frac_bin = factor(frac_bin, levels = bin_levels),
    frac_plot = case_when(
      contrast_type == "Spatially adjusted" ~ round(frac_area, 6),
      contrast_type == "Current landscape" ~ as.numeric(frac_bin)
    ),
    response = response_var
  )
  
  contrasts_summary <- contrasts_long %>%
    filter(!is.na(frac_plot), !is.na(difference)) %>%
    group_by(region, response, contrast_type, frac_plot, frac_bin, stage) %>%
    summarise(
      median = median(difference, na.rm = TRUE),
      lower = safe_quantile(difference, 0.025),
      upper = safe_quantile(difference, 0.975),
      n_runs = n_distinct(run),
      .groups = "drop"
    )
  
  fwrite(
  contrasts_summary,
  file.path(out_dir, "tables", "contrasts_summary_empirical_CI.csv"),  sep = ";",  dec = ",")
  
  year_summary <- year_effects_ensemble %>%
    mutate(response = response_var) %>%
    group_by(region, response, Ano) %>%
    summarise(
      median = median(year_effect, na.rm = TRUE),
      lower = safe_quantile(year_effect, 0.025),
      upper = safe_quantile(year_effect, 0.975),
      n_runs = sum(!is.na(year_effect)),
      .groups = "drop"
    )
  
  fwrite(
  year_summary,
  file.path(out_dir, "tables", "year_effects_summary_empirical_CI.csv"),  sep = ";",  dec = ",")
  
  spatial_summary <- spatial_component_ensemble %>%
    mutate(response = response_var) %>%
    group_by(region, response, Lon, Lat) %>%
    summarise(
      spatial_median = median(spatial_component, na.rm = TRUE),
      spatial_mean = mean(spatial_component, na.rm = TRUE),
      spatial_sd = sd(spatial_component, na.rm = TRUE),
      n_runs = n(),
      .groups = "drop"
    )
  
  fwrite(
  spatial_summary,
  file.path(out_dir, "tables", "spatial_component_summary.csv"),  sep = ";",  dec = ",")
  
  metrics_summary <- metrics_ensemble %>%
    group_by(region, response) %>%
    summarise(
      n_successful_runs = n(),
      
      CCC_median = median(CCC, na.rm = TRUE),
      CCC_lower = safe_quantile(CCC, 0.025),
      CCC_upper = safe_quantile(CCC, 0.975),
      
      RMSE_median = median(RMSE, na.rm = TRUE),
      RMSE_lower = safe_quantile(RMSE, 0.025),
      RMSE_upper = safe_quantile(RMSE, 0.975),
      
      MAE_median = median(MAE, na.rm = TRUE),
      MAE_lower = safe_quantile(MAE, 0.025),
      MAE_upper = safe_quantile(MAE, 0.975),
      
      Bias_median = median(Bias, na.rm = TRUE),
      Bias_lower = safe_quantile(Bias, 0.025),
      Bias_upper = safe_quantile(Bias, 0.975),
      
      R2_median = median(R2, na.rm = TRUE),
      R2_lower = safe_quantile(R2, 0.025),
      R2_upper = safe_quantile(R2, 0.975),
      
      .groups = "drop"
    )

  fwrite(
    metrics_summary,
    file.path(out_dir, "tables", "metrics_summary_empirical_CI.csv"),  sep = ";",  dec = ",")

  
  list(
    contrasts_summary = contrasts_summary,
    year_summary = year_summary,
    spatial_summary = spatial_summary
  )
}

load_diag_results <- function(response_var, region_name) {
  
  out_dir <- get_out_dir(response_var, region_name)
  
  diagnostics_ensemble <- read_all_files(
    file.path(out_dir, "diagnostics"),

  )
  
  metrics_ensemble <- read_all_files(
    file.path(out_dir, "metrics"),
    "^metrics_run_.*\\.csv$"
  )
  
  if (nrow(diagnostics_ensemble) == 0) {
    stop("No diagnostics files found for ", response_var, " in ", out_dir)
  }
  
  if (nrow(metrics_ensemble) == 0) {
    stop("No metrics files found for ", response_var, " in ", out_dir)
  }

    mutate(
      response = response_var,
      variable_label = response_labels[[response_var]]
    )
  
  metrics_ensemble <- metrics_ensemble %>%
    mutate(
      response = response_var,
      variable_label = response_labels[[response_var]]
    )
  
  list(
    diagnostics = diagnostics_ensemble,
    metrics = metrics_ensemble
  )
}

read_all_files <- function(path, pattern) {
  files <- list.files(path, pattern = pattern, full.names = TRUE)
  if (length(files) == 0) return(data.table())
  rbindlist(lapply(files, fread), fill = TRUE)
}

load_diag_results <- function(response_var, region_name) {
  
  out_dir <- get_out_dir(response_var, region_name)
  
  diagnostics_ensemble <- read_all_files(
    file.path(out_dir, "diagnostics"),
    "^diagnostics_run_.*\\.csv$"
  )
  
  metrics_ensemble <- read_all_files(
    file.path(out_dir, "metrics"),
    "^metrics_run_.*\\.csv$"
  )
  
  if (nrow(diagnostics_ensemble) == 0) {
    stop("No diagnostics files found for ", response_var, " in ", out_dir)
  }
  
  if (nrow(metrics_ensemble) == 0) {
    stop("No metrics files found for ", response_var, " in ", out_dir)
  }
  
  diagnostics_ensemble <- diagnostics_ensemble %>%
    mutate(
      response = response_var,
      variable_label = response_labels[[response_var]]
    )
  
  metrics_ensemble <- metrics_ensemble %>%
    mutate(
      response = response_var,
      variable_label = response_labels[[response_var]]
    )
  
  list(
    diagnostics = diagnostics_ensemble,
    metrics = metrics_ensemble
  )
}

sample_diag <- function(dat, n_max = 30000) {
  set.seed(123)
